Rogue-pod

apiVersion: v1
kind: Pod
metadata:
  name: rogue-pod
spec:
  containers:
  - name: snooper-container
    image: busybox
    command: ["sh", "-c", "while true; do cat /data/secret.txt; sleep 3; done"]
    volumeMounts:
    - name: data-vol
      mountPath: /data
  volumes:
  - name: data-vol
    hostPath:
      path: /opt/sensitive-data
~

catch it by Falco rule:

- rule: Catch Sensitive Directory Access
  desc: Detects when a container accesses files in our target path
  condition: fd.name contains "secret.txt" and container.id != "host"
  output: "BINGO! Rogue container found -> Name: %container.name | ID: %container.id | File: %fd.name | Cmd: %proc.cmdline"
  priority: WARNING

run: "falco -r catch-rogue.yaml -M 15"
